cmake_minimum_required(VERSION 3.14)
project(GymgymClient)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(ftxui GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git GIT_TAG v5.0.0)
FetchContent_MakeAvailable(ftxui)
add_executable(GymgymClient src/main.cpp)
target_link_libraries(GymgymClient PRIVATE ftxui::component ftxui::dom ftxui::screen)
if(WIN32)
target_link_libraries(GymgymClient PRIVATE ws2_32)
endif()
